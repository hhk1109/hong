================================================================================
一、建库语句
================================================================================

CREATE DATABASE IF NOT EXISTS minion_db
  DEFAULT CHARACTER SET utf8mb4
  DEFAULT COLLATE utf8mb4_unicode_ci;

USE minion_db;

================================================================================
二、建表语句
================================================================================

CREATE TABLE IF NOT EXISTS `user` (
  `id`         BIGINT       NOT NULL AUTO_INCREMENT COMMENT '主键',
  `username`   VARCHAR(13)  NOT NULL                COMMENT '账号（11~13位数字）',
  `password`   VARCHAR(100) NOT NULL                COMMENT '密码（BCrypt哈希）',
  `created_at` DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_username` (`username`)
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4 COMMENT = '小黄人用户表';

================================================================================
三、字段说明
================================================================================

  字段名       类型          是否可空   说明
  -----------  ------------  ---------  --------------------------------------
  id           BIGINT        否         主键，自增
  username     VARCHAR(13)   否         账号，11~13 位纯数字，唯一索引
  password     VARCHAR(100)  否         密码，BCrypt 哈希（$2a$10$ 开头，非明文）
  created_at   DATETIME      否         创建时间，插入时自动填充
  updated_at   DATETIME      否         更新时间，记录变更时自动刷新

  索引说明：
    PRIMARY KEY (`id`)                     主键索引
    UNIQUE KEY  `uk_username` (`username`) 账号唯一索引（重复注册由它兜底拦截）

================================================================================
五、注意事项
================================================================================

  1. 表名 user 使用了反引号包裹：user 是 MySQL 保留字附近的常见词，反引号可避免歧义。
  2. 字符集统一 utf8mb4，避免中文/emoji 存储乱码。
  3. 账号唯一性由数据库唯一索引 + 后端注册前查询双重保证。
  4. 密码字段长度 100 是为 BCrypt（固定 60 位）预留余量。
  5. 生产环境建议：数据库账号不要用 root；密码不要硬编码在 application.yml，
     应改为环境变量或配置中心注入。
